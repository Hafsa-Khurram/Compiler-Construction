# This is a BeginnerScript sample program

store username;
store age;

# Taking input from user
input "Enter your name: " into username;
input "Enter your age: " into age;

# Displaying a welcome message
display ["Hello, " + username];

# Checking eligibility
if age >= 18 then [
  display ["You are eligible to vote."];
] otherwise [
  display ["You are not eligible to vote yet."];
]

# Defining a function
function welcome() [
  display ["Welcome to BeginnerScript!"];
]

# Calling the function
welcome();

# Using repeat loop
repeat 3 times [
  display ["Thank you for using BeginnerScript, " + username];
]

# End of program
